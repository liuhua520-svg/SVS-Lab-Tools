# requirements-qwen3tts.txt
# qwen3tts_server.py 独立服务依赖（不要装进主 .mfa_env，也不要和
# .qwen3_env / .nemo_env 共用！）
#
# 用法：
#   conda create --prefix ../.qwen3tts_env python=3.12 -y
#   conda activate ../.qwen3tts_env
#   pip install -r requirements-qwen3tts.txt
#   python qwen3tts_server.py
#
# 说明：
#   这是 Qwen3-TTS（CustomVoice / VoiceDesign / Base-VoiceClone 三套
#   12Hz checkpoint）跑在独立进程里所需的全部依赖。主 Flask 应用
#   （app.py 所在的 .mfa_env）只通过 HTTP 调用 127.0.0.1:5003，完全不
#   需要装 qwen-tts / 对应版本的 transformers 这一套。
#
#   之所以单独建环境：qwen-tts 官方要求较新的 transformers 版本（用于
#   支持 Qwen3-TTS-Tokenizer-12Hz 的模型结构），与 .mfa_env 里 qwen-asr
#   锁定的 transformers>=4.40.0 不一定是同一个精确版本，装进同一个 venv
#   容易互相"打架"，把对方需要的版本降级。
#
#   官方推荐 Python 3.12（见 https://github.com/QwenLM/Qwen3-TTS
#   Environment Setup 一节），这里与之保持一致。

# 【修复】原先 --index-url 写在文件末尾 torch 那几行旁边（此前还曾错误
# 地写成 --extra-index-url + cu118 放在顶部、但下面激活的是 +cu121，两处
# 不一致）。这里统一为一条规则：pip 对 requirements 文件里的
# --index-url/--extra-index-url 是"全局生效"的，不是只作用于紧邻的那几
# 行。用 --index-url 会把默认 PyPI 源整个替换掉，导致同一文件里的
# flask / qwen-tts / transformers 等其它包全部只在
# download.pytorch.org/whl/cpu（只托管 PyTorch 系列 wheel）里找，报
# "Could not find a version that satisfies the requirement flask==2.3.3
# (from versions: none)"。改为 --extra-index-url 并放在文件顶部：追加而
# 非替换默认源，PyPI 包正常从 PyPI 装，torch 系列在 PyPI 找不到时才去
# PyTorch 官方索引找，与 requirements.txt / requirements-nemo.txt /
# requirements-qwen3.txt 三个文件的写法保持一致。
--extra-index-url https://download.pytorch.org/whl/cpu

# ====================== 服务框架 ======================
flask==2.3.3

# ====================== Qwen3-TTS ======================
# 官方包名为 qwen-tts（import 时是 qwen_tts），见
# https://github.com/QwenLM/Qwen3-TTS
qwen-tts==0.1.1
# qwen-tts 自身在 PyPI 上精确锁定了 transformers==4.57.3 /
# accelerate==1.12.0（见其 requires_dist，注意与 requirements-qwen3.txt
# 里 qwen-asr 精确锁定的 transformers==4.57.6 是不同的精确版本——这正是
# 上面说明里强调必须与 .qwen3_env 分开建环境的原因，两边各自的精确锁定
# 装进同一个 venv 会互相冲突）。下面两行不加版本号，交给 pip 解析时以
# qwen-tts 的精确锁定为准；这里显式列出只是为了文档留痕，方便日后追踪
# 版本变化，不会影响实际安装到的版本。
transformers==4.57.3
accelerate==1.12.0
# ====================== 基础数值库 ======================
# 【修复】原先这里写的是 numpy=1.26.4（单等号），不是合法的 pip 版本
# 约束语法（PEP 440 要求 ==/>=/<=/!=/~=/>/< 这些双字符或单字符比较符，
# 单独一个 "=" 不在其中），会导致 pip 直接报
# "Invalid requirement: 'numpy=1.26.4'" 而整个安装失败。改为合法的
# "==" 精确锁定，与其它三个 requirements 文件的 numpy==1.26.4 保持一致
# （qwen-tts 自身对 numpy 无显式版本要求，锁 1.26.4 不会有冲突）。
numpy==1.26.4

# ====================== 音频处理 ======================
soundfile>=0.12.1

# ====================== 网络与其他基础库 ======================
requests
tqdm

# ====================== （可选）FlashAttention 2 ======================
# 官方建议安装以降低显存占用（仅在 GPU + torch.float16/bfloat16 下生效，
# 未安装时 qwen3tts_server.py 会自动跳过、回退到默认 attention 实现，
# 不影响功能，只是速度/显存表现略差）。按官方文档安装较为繁琐（需要
# --no-build-isolation，且如内存/CPU 核心有限需加 MAX_JOBS=4），因此
# 这里注释掉、按需手动安装：
#   pip install -U flash-attn --no-build-isolation
# 详见 https://github.com/Dao-AILab/flash-attention

# ====================== PyTorch (根据硬件三选一，与主项目一致) ======================
# 注意：CUDA 版 wheel（+cu121 / +cu118）不在 PyPI 默认源上，只在文件顶部
# 那行 --extra-index-url 指向的 PyTorch 官方索引里。若切换到 GPU 选项，
# 顶部的 --extra-index-url 需要改成对应 CUDA 版本的索引地址（cu121 或
# cu118），不要再在这里逐个选项各写一条 --index-url。

# CUDA 12.1（用于 RTX 30/40 系等现代显卡）
# torch==2.3.1+cu121
# torchvision==0.18.1+cu121
# torchaudio==2.3.1+cu121

# CUDA 11.8（旧版显卡保留选项，如 GTX10系显卡 等 Pascal 架构）
# torch==2.3.1+cu118
# torchvision==0.18.1+cu118
# torchaudio==2.3.1+cu118

# CPU Only（无显卡，默认激活）
torch==2.3.1+cpu
torchvision==0.18.1+cpu
torchaudio==2.3.1+cpu
